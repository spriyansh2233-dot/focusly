);

-- Concepts/Topics
CREATE TABLE concepts (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(255) NOT NULL,
    subject VARCHAR(100) NOT NULL,
    description TEXT,
    difficulty_level DECIMAL(3,2), -- 0.0 to 1.0
    prerequisites JSONB, -- array of prerequisite concept IDs
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Learning Paths
CREATE TABLE learning_paths (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    goal_description TEXT,
    pathway JSONB NOT NULL, -- graph structure of concepts
    current_week INT DEFAULT 1,
    target_completion_date DATE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- User Progress Tracking
CREATE TABLE user_progress (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    concept_id UUID REFERENCES concepts(id) ON DELETE CASCADE,
    mastery_level VARCHAR(20) CHECK (mastery_level IN ('learning', 'reviewing', 'mastered')),
    last_reviewed_at TIMESTAMP,
    next_review_at TIMESTAMP,
    review_count INT DEFAULT 0,
    ease_factor DECIMAL(3,2) DEFAULT 2.5, -- for spaced repetition
    interval_days INT DEFAULT 1,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(user_id, concept_id)
);

-- Quiz Questions
CREATE TABLE quiz_questions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

3. Service Layer
java@Service
@Transactional
public class LearningPathService {
    
    @Autowired
    private UserProgressRepository progressRepository;
    
    @Autowired
    private ConceptRepository conceptRepository;
    
    @Autowired
    private SpacedRepetitionService spacedRepService;
    
    /**
     * Generate personalized learning path based on user assessment
     */
    public LearningPath generateLearningPath(UUID userId, String subject, 
                                             AssessmentResult assessment) {
        // Get all concepts for subject
        List<Concept> allConcepts = conceptRepository.findBySubject(subject);
        
        // Filter out already mastered concepts
        Set<UUID> masteredIds = progressRepository.findMasteredConcepts(userId)
            .stream()
            .map(up -> up.getConcept().getId())
            .collect(Collectors.toSet());
        
        List<Concept> remainingConcepts = allConcepts.stream()
            .filter(c -> !masteredIds.contains(c.getId()))
            .collect(Collectors.toList());
        
        // Sort by prerequisites and difficulty
        List<Concept> orderedPath = topologicalSort(remainingConcepts);
        
        // Create weekly milestones
        LearningPath path = new LearningPath();
        path.setUserId(userId);
        path.setPathway(buildPathwayJson(orderedPath));
        
        return learningPathRepository.save(path);
    }
    
    /**
     * Update progress after quiz attempt
     */
    public void updateProgress(UUID userId, UUID conceptId, boolean isCorrect) {
        UserProgress progress = progressRepository
            .findByUserIdAndConceptId(userId, conceptId)
            .orElseGet(() -> createNewProgress(userId, conceptId));
        
        // Apply spaced repetition algorithm
        SpacedRepetitionResult result = spacedRepService
            .calculateNextReview(progress, isCorrect);
        
        progress.setEaseFactor(result.getEaseFactor());
        progress.setIntervalDays(result.getIntervalDays());
        progress.setNextReviewAt(result.getNextReviewDate());
        progress.setReviewCount(progress.getReviewCount() + 1);
        
        // Update mastery level
        if (progress.getReviewCount() >= 5 && result.getIntervalDays() >= 30) {
            progress.setMasteryLevel(MasteryLevel.MASTERED);
        } else if (progress.getReviewCount() >= 2) {
            progress.setMasteryLevel(MasteryLevel.REVIEWING);
        }
        
        progressRepository.save(progress);
    }
}

4. AI Service Integration
java@Service
public class AITutorService {
    
    @Value("${anthropic.api.key}")
    private String anthropicApiKey;
    
    private final WebClient webClient;
    
    public AITutorService(WebClient.Builder webClientBuilder) {
        this.webClient = webClientBuilder
            .baseUrl("https://api.anthropic.com")
            .build();
    }
    
    /**
     * Get AI explanation for a concept
     */
    public String explainConcept(String concept, String learningStyle, 
                                 String userLevel) {
        
        String prompt = buildExplanationPrompt(concept, learningStyle, userLevel);
        
        ClaudeRequest request = ClaudeRequest.builder()
            .model("claude-sonnet-4-20250514")
            .maxTokens(1024)
            .messages(List.of(
                new Message("user", prompt)
            ))
            .build();
        
        ClaudeResponse response = webClient.post()
            .uri("/v1/messages")
            .header("x-api-key", anthropicApiKey)
            .header("anthropic-version", "2023-06-01")
            .bodyValue(request)
            .retrieve()
            .bodyToMono(ClaudeResponse.class)
            .block();
        
        return response.getContent().get(0).getText();
    }
    
    /**
     * Generate adaptive quiz questions
     */
    public List<QuizQuestion> generateQuiz(UUID conceptId, double difficulty) {
        Concept concept = conceptRepository.findById(conceptId)
            .orElseThrow(() -> new ResourceNotFoundException("Concept not found"));
        
        String prompt = String.format(
            "Generate 5 %s-level quiz questions about %s. " +
            "Return as JSON array with fields: question, options, correctAnswer, explanation",
            getDifficultyLabel(difficulty),
            concept.getName()
        );
        
        String aiResponse = callClaudeAPI(prompt);
        
        // Parse JSON and convert to QuizQuestion entities
        return parseQuizQuestionsFromJson(aiResponse, conceptId);
    }
    
    private String buildExplanationPrompt(String concept, String style, 
                                         String level) {
        return String.format(
            "Explain %s to a %s student. " +
            "Use a %s learning approach (e.g., %s). " +
            "Include an analogy and a practical example.",
            concept,
            level,
            style,
            getStyleGuidance(style)
        );
    }
}

5. Spaced Repetition Algorithm
java@Service
public class SpacedRepetitionService {
    
    /**
     * SM-2 Algorithm implementation
     * Returns next review date and updated ease factor
     */
    public SpacedRepetitionResult calculateNextReview(
        UserProgress progress, boolean isCorrect) {
        
        double easeFactor = progress.getEaseFactor();
        int interval = progress.getIntervalDays();
        
        if (isCorrect) {
            // Increase interval
            if (interval == 1) {
                interval = 6;
            } else {
                interval = (int) Math.round(interval * easeFactor);
            }
            
            // Adjust ease factor slightly up
            easeFactor = Math.min(2.5, easeFactor + 0.1);
            
        } else {
            // Reset to beginning
            interval = 1;
            easeFactor = Math.max(1.3, easeFactor - 0.2);
        }
        
        LocalDateTime nextReview = LocalDateTime.now().plusDays(interval);
        
        return SpacedRepetitionResult.builder()
            .easeFactor(easeFactor)
            .intervalDays(interval)
            .nextReviewDate(nextReview)
            .build();
    }
}

     * Get today's review quiz for user
     */

        return ResponseEntity.ok(

                .questions(questions)
                .totalQuestions(questions.size())
                .build()
        );
    }
    
    /**
     * Submit quiz answer
     */
    @PostMapping("/submit")
    public ResponseEntity<SubmissionResult> submitAnswer(
        @AuthenticationPrincipal User user,
        @Valid @RequestBody QuizSubmission submission) {
        
        boolean isCorrect = quizService.checkAnswer(
            submission.getQuestionId(),
            submission.getUserAnswer()
        );
        
        // Update progress with spaced repetition
        learningPathService.updateProgress(
            user.getId(),
            submission.getConceptId(),
            isCorrect
        );
        
        return ResponseEntity.ok(
            SubmissionResult.builder()
                .isCorrect(isCorrect)
                .explanation(quizService.getExplanation(submission.getQuestionId()))
                .nextReviewIn(quizService.getNextReviewInterval(user.getId()))
                .build()
        );
    }
}

                .requestMatchers("/api/v1/**").authenticated()
            )
            .sessionManagement(session -> session
                .sessionCreationPolicy(SessionCreationPolicy.STATELESS)
            )
            .addFilterBefore(jwtAuthenticationFilter(), 
                UsernamePasswordAuthenticationFilter.class);
        
        return http.build();
    }
    
    @Bean
    public JwtAuthenticationFilter jwtAuthenticationFilter() {
        return new JwtAuthenticationFilter();
    }
}

@Component
public class JwtTokenProvider {
    
    @Value("${jwt.secret}")
    private String jwtSecret;
    
    public String generateToken(User user) {
        return Jwts.builder()
            .setSubject(user.getId().toString())
            .setIssuedAt(new Date())
            .setExpiration(new Date(System.currentTimeMillis() + 86400000)) // 24h
            .signWith(SignatureAlgorithm.HS512, jwtSecret)
            .compact();
    }
    
    public UUID getUserIdFromToken(String token) {
        Claims claims = Jwts.parser()
            .setSigningKey(jwtSecret)
            .parseClaimsJws(token)
            .getBody();
        
        return UUID.fromString(claims.getSubject());
    }
}
